# Spotify API Configuration
# Para usar a API oficial do Spotify, você precisa:
# 1. Criar uma conta de desenvolvedor no Spotify: https://developer.spotify.com/
# 2. Criar uma aplicação e obter Client ID e Client Secret
# 3. Configurar as variáveis abaixo

VITE_SPOTIFY_CLIENT_ID=your_spotify_client_id_here
VITE_SPOTIFY_CLIENT_SECRET=your_spotify_client_secret_here

# API Base URL (opcional - para fallback)
VITE_API_URL=http://localhost:3001
